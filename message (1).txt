local function New(class, props, children)
	local inst = Instance.new(class)
	for k, v in props or {} do
		if k ~= "Parent" then
			inst[k] = v
		end
	end
	for _, child in children or {} do
		child.Parent = inst
	end
	inst.Parent = props and props.Parent or nil
	return inst
end

local function Fire(callback, ...)
	if type(callback) ~= "function" then
		return
	end
	local args = table.pack(...)
	task.spawn(function()
		local ok, err = pcall(callback, table.unpack(args, 1, args.n))
		if not ok then
			warn("[UI] callback error: " .. tostring(err))
		end
	end)
end

local function NextOrder(container)
	container._order = (container._order or 0) + 1
	return container._order
end

local function IsPress(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

local function IsMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

local function MakeDraggable(hit, scroll, callback)
	local dragging = false

	local function update(input)
		local pos, size = hit.AbsolutePosition, hit.AbsoluteSize
		if size.X <= 0 or size.Y <= 0 then
			return
		end
		callback(
			math.clamp((input.Position.X - pos.X) / size.X, 0, 1),
			math.clamp((input.Position.Y - pos.Y) / size.Y, 0, 1)
		)
	end

	hit.InputBegan:Connect(function(input)
		if IsPress(input) then
			dragging = true
			if scroll then
				scroll.ScrollingEnabled = false
			end
			update(input)
		end
	end)
	Connect(UIS.InputChanged, function(input)
		if dragging and IsMove(input) then
			update(input)
		end
	end)
	Connect(UIS.InputEnded, function(input)
		if dragging and IsPress(input) then
			dragging = false
			if scroll then
				scroll.ScrollingEnabled = true
			end
		end
	end)
end

local function EnsureFolders()
	if typeof(makefolder) ~= "function" or typeof(isfolder) ~= "function" then
		return false
	end
	for _, path in { "Mal", CONFIG_FOLDER } do
		if not isfolder(path) then
			pcall(makefolder, path)
		end
	end
	return true
end

local function ReadCache(path)
	if typeof(isfile) ~= "function" or typeof(readfile) ~= "function" then
		return nil
	end
	local ok, data = pcall(function()
		if isfile(path) then
			return readfile(path)
		end
	end)
	if ok and type(data) == "string" and #data > 0 then
		return data
	end
	return nil
end

local function WriteCache(path, data)
	if typeof(writefile) ~= "function" or not EnsureFolders() then
		return false
	end
	return (pcall(writefile, path, data))
end

local Molto = nil
local IconsReady = false
local IconMeta = {}
local MetaDirty = false

local function RunMolto(source)
	if type(source) ~= "string" or #source < 50 then
		return nil
	end
	local ok, result = pcall(function()
		return assert(loadstring(source))()
	end)
	if ok and type(result) == "table" then
		return result
	end
	return nil
end

local function LoadIcons(preferOffline)
  if IconsReady then
      return
  end
  IconsReady = true

  local metaRaw = ReadCache(ICON_CACHE)
  if metaRaw then
      local ok, decoded = pcall(function()
          return HttpService:JSONDecode(metaRaw)
      end)

      if ok and type(decoded) == "table" then
          IconMeta = decoded
      end
  end

  local ok, body = pcall(function()
      return game:HttpGet(MOLTO_CUSTOM_URL)
  end)

  if ok and type(body) == "string" then
      Molto = RunMolto(body)

      if Molto then
          WriteCache(MOLTO_CUSTOM_CACHE, body)
      end
  end

  if not Molto then
      Molto = RunMolto(ReadCache(MOLTO_CUSTOM_CACHE))
  end

  if not Molto then
      local okOfficial, official = pcall(function()
          return game:HttpGet(MOLTO_URL)
      end)

      if okOfficial and type(official) == "string" then
          Molto = RunMolto(official)
      end
  end

  if not Molto then
      warn("[UI] Molto unavailable (hosted, cached and official), no icons")
      return
  end

  pcall(function()
      Molto.PreloadAsync()
      Molto.PreloadAsync({ "x-bold", "maximize", "minimize", "minus", "menu", "wine", "backpack" })
  end)

  pcall(function()
      Molto.PreloadAsync()
  end)
end

end

local function ResolveAsset(name)
	if Molto then
		local ok, asset = pcall(Molto.GetAsset, name)
		if ok and asset then
			pcall(function()
				local meta = {}
				if asset.FontFace and asset.Text then
					meta.Text = asset.Text
					meta.Family = asset.FontFace.Family
					meta.Weight = asset.FontFace.Weight.Name
					meta.Style = asset.FontFace.Style.Name
				else
					meta.Url = asset.Url
					meta.RS = { asset.ImageRectSize.X, asset.ImageRectSize.Y }
					meta.RO = { asset.ImageRectOffset.X, asset.ImageRectOffset.Y }
				end
				local old = IconMeta[name]
				if not old or old.Family ~= meta.Family or old.Url ~= meta.Url or old.Text ~= meta.Text then
					IconMeta[name] = meta
					MetaDirty = true
				end
			end)
			return asset
		end
	end

	local meta = IconMeta[name]
	if type(meta) == "table" then
		if meta.Text and meta.Family then
			local ok, face = pcall(function()
				return Font.new(meta.Family, Enum.FontWeight[meta.Weight], Enum.FontStyle[meta.Style])
			end)
			if ok and face then
				return { FontFace = face, Text = meta.Text }
			end
		elseif meta.Url and meta.RS and meta.RO then
			return {
				Url = meta.Url,
				ImageRectSize = Vector2.new(meta.RS[1], meta.RS[2]),
				ImageRectOffset = Vector2.new(meta.RO[1], meta.RO[2]),
			}
		end
	end
	return nil
end

local function CreateIcon(parent, name, size, color)
	local holder = New("Frame", {
		Name = "Icon",
		BackgroundTransparency = 1,
		Size = UDim2.fromOffset(size, size),
		Parent = parent,
	})
	local icon = { Instance = holder, Color = color, Name = name, HasGlyph = false }
	local inner = nil

	function icon:Set(newName)
		self.Name = newName
		self.HasGlyph = false
		if inner then
			inner:Destroy()
			inner = nil
		end
		if not newName or newName == "" then
			return
		end

		if tostring(newName):match("^rbxasset") or tostring(newName):match("^%d+$") then
			local id = tostring(newName):match("^%d+$") and ("rbxassetid://" .. newName) or newName
			inner = New("ImageLabel", {
				BackgroundTransparency = 1,
				Size = UDim2.fromScale(1, 1),
				Image = id,
				ImageColor3 = self.Color,
				Parent = holder,
			})
			self.HasGlyph = true
			return
		end

		local asset = ResolveAsset(newName)
		if not asset then
			return
		end

		if asset.FontFace and asset.Text then
			inner = New("TextLabel", {
				BackgroundTransparency = 1,
				Size = UDim2.fromScale(1, 1),
				Text = asset.Text,
				FontFace = asset.FontFace,
				TextSize = math.min(size, 100),
				TextColor3 = self.Color,
				Parent = holder,
			})
		else
			inner = New("ImageLabel", {
				BackgroundTransparency = 1,
				Size = UDim2.fromScale(1, 1),
				Image = asset.Url,
				ImageRectSize = asset.ImageRectSize,
				ImageRectOffset = asset.ImageRectOffset,
				ImageColor3 = self.Color,
				Parent = holder,
			})
		end
		self.HasGlyph = true
	end

	function icon:SetColor(newColor, duration)
		self.Color = newColor
		if inner then
			local prop = inner:IsA("TextLabel") and "TextColor3" or "ImageColor3"
			if duration then
				Tween(inner, duration, { [prop] = newColor })
			else
				inner[prop] = newColor
			end
		end
	end

	icon:Set(name)
	return icon
end

local function Encode(value)
	local kind = typeof(value)
	if kind == "Color3" then
		return { __t = "Color3", Hex = value:ToHex() }
	elseif kind == "EnumItem" then
		return { __t = "KeyCode", Name = value.Name }
	end
	return value
end

local function Decode(value)
	if type(value) == "table" and value.__t == "Color3" then
		return Color3.fromHex(value.Hex)
	elseif type(value) == "table" and value.__t == "KeyCode" then
		local ok, key = pcall(function()
			return Enum.KeyCode[value.Name]
		end)
		return ok and key or Enum.KeyCode.Unknown
	end
	return value
end

function Library:SaveConfig(name)
	assert(typeof(writefile) == "function" and EnsureFolders(), "File functions are not available in this executor")
	local data = {}
	for flag, value in self.Flags do
		data[flag] = Encode(value)
	end
	writefile(`{CONFIG_FOLDER}/{name}.json`, HttpService:JSONEncode(data))
end

function Library:LoadConfig(name)
	assert(typeof(readfile) == "function" and typeof(isfile) == "function", "File functions are not available")
	local path = `{CONFIG_FOLDER}/{name}.json`
	assert(isfile(path), `Config "{name}" does not exist`)
	local data = HttpService:JSONDecode(readfile(path))
	for flag, raw in data do
		local option = self.Options[flag]
		if option then
			pcall(function()
				option:Set(Decode(raw))
			end)
		end
	end
end

function Library:DeleteConfig(name)
	local path = `{CONFIG_FOLDER}/{name}.json`
	if typeof(isfile) == "function" and typeof(delfile) == "function" and isfile(path) then
		delfile(path)
	end
end

function Library:GetConfigs()
	local names = {}
	if typeof(listfiles) == "function" and EnsureFolders() then
		for _, path in listfiles(CONFIG_FOLDER) do
			local file = path:gsub("\\", "/"):match("([^/]+)$")
			local name = file and file:match("^(.+)%.json$")
			if name then
				table.insert(names, name)
			end
		end
	end
	table.sort(names)
	return names
end

local function KillToggle(toggle)
	if toggle._Dead or not toggle.Value then
		return
	end
	toggle.Value = false
	if toggle.Flag then
		Library.Flags[toggle.Flag] = false
	end
	local callback = toggle._Callback
	if type(callback) == "function" then
		task.spawn(function()
			local ok, err = pcall(callback, false)
			if not ok then
				warn("[UI] unload callback error: " .. tostring(err))
			end
		end)
	end
end

local function Finish(obj, row, titleLabel)
	obj.Row = row
	function obj:SetVisible(state)
		row.Visible = state and true or false
	end
	function obj:SetTitle(text)
		if titleLabel then
			titleLabel.Text = text
		end
	end
	function obj:Destroy()
		if self.Type == "Toggle" then
			KillToggle(self)
			self._Dead = true
			local index = table.find(Library._Toggles, self)
			if index then
				table.remove(Library._Toggles, index)
			end
		end
		row:Destroy()
		if self.Flag then
			Library.Options[self.Flag] = nil
			Library.Flags[self.Flag] = nil
		end
	end
	return obj
end

function Elements:Slider(opts)
	opts = opts or {}
	local T = Library.Theme
	local min, max = opts.Min or 0, opts.Max or 100
	local increment = opts.Increment or 1
	local suffix = opts.Suffix or ""

	local row, header, title = NewRow(self, opts.Name or "Slider", opts.Description, 70)

	local valueBox = New("TextBox", {
		Name = "Value",
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, 0, 0.5, 0),
		Size = UDim2.fromOffset(70, 26),
		BackgroundColor3 = T.Surface,
		Font = Fonts.Medium,
		TextSize = 12,
		TextColor3 = T.Text,
		ClearTextOnFocus = false,
		ZIndex = 4,
		Parent = header,
	})
	Corner(valueBox, 7)
	Stroke(valueBox, T.Stroke, 1, 0.3)

	local area = New("Frame", {
		Name = "SliderArea",
		BackgroundTransparency = 1,
		Size = UDim2.new(1, 0, 0, 30),
		LayoutOrder = 2,
		Parent = row,
	})
	local track = New("Frame", {
		AnchorPoint = Vector2.new(0, 0.5),
		Position = UDim2.new(0, 0, 0.35, 0),
		Size = UDim2.new(1, 0, 0, 6),
		BackgroundColor3 = T.Off,
		Parent = area,
	})
	Corner(track, 3)
	local fill = New("Frame", {
		Size = UDim2.fromScale(0, 1),
		BackgroundColor3 = T.Accent,
		Parent = track,
	})
	Corner(fill, 3)
	New("UIGradient", { Color = ColorSequence.new(T.AccentDark, T.AccentLight), Parent = fill })
	local knob = New("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0, 0.5),
		Size = UDim2.fromOffset(16, 16),
		BackgroundColor3 = T.Text,
		Parent = track,
	})
	Corner(knob, 8)
	Stroke(knob, T.Accent, 2, 0)

	local hit = New("TextButton", {
		Name = "Hit",
		Text = "",
		AutoButtonColor = false,
		BackgroundTransparency = 1,
		Size = UDim2.fromScale(1, 1),
		ZIndex = 3,
		Parent = area,
	})

	local function snap(value)
		value = math.clamp(value, min, max)
		value = math.floor((value - min) / increment + 0.5) * increment + min
		return math.clamp(tonumber(string.format("%.4f", value)) or min, min, max)
	end

	local function format(value)
		local text = string.format("%.3f", value)
		text = text:gsub("0+$", "")
		text = text:gsub("%.$", "")
		return text .. suffix
	end

	local obj = { Type = "Slider", Value = min }

	local function render()
		local alpha = (max == min) and 0 or (obj.Value - min) / (max - min)
		fill.Size = UDim2.fromScale(alpha, 1)
		knob.Position = UDim2.fromScale(alpha, 0.5)
		valueBox.Text = format(obj.Value)
	end

	function obj:Set(value, silent)
		value = tonumber(value)
		if not value then
			render()
			return
		end
		value = snap(value)
		local changed = value ~= self.Value
		self.Value = value
		render()
		Sync(self)
		if changed and not silent then
			Fire(opts.Callback, value)
		end
	end
	function obj:Get()
		return self.Value
	end

	MakeDraggable(hit, self._Ctx.Scroll, function(x)
		obj:Set(min + (max - min) * x)
	end)
	valueBox.FocusLost:Connect(function()
		local typed = tonumber((valueBox.Text:gsub("[^%d%.%-]", "")))
		if typed then
			obj:Set(typed)
		else
			render()
		end
	end)

	obj.Value = snap(opts.Default or min)
	render()
	Register(obj, opts)
	return Finish(obj, row, title)
end

function Elements:Dropdown(opts)
	opts = opts or {}
	local T = Library.Theme
	local ctrlW = IsMobile and 122 or 160
	local row, header, title = NewRow(self, opts.Name or "Dropdown", opts.Description, ctrlW)
	local click = ClickArea(row, header)

	local box = New("Frame", {
		Name = "Box",
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, 0, 0.5, 0),
		Size = UDim2.fromOffset(ctrlW, 30),
		BackgroundColor3 = T.Surface,
		ZIndex = 4,
		Parent = header,
	})
	Corner(box, 8)
	local boxStroke = Stroke(box, T.Stroke, 1, 0.2)

	local valueLabel = New("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.fromOffset(10, 0),
		Size = UDim2.new(1, -34, 1, 0),
		Font = Fonts.Medium,
		TextSize = 13,
		TextColor3 = T.Text,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
		Text = "",
		Parent = box,
	})
	local chevron = CreateIcon(box, "chevron-down", 16, T.SubText)
	chevron.Instance.AnchorPoint = Vector2.new(1, 0.5)
	chevron.Instance.Position = UDim2.new(1, -8, 0.5, 0)

	local wrapper = NewWrapper(row)
	local drop = New("Frame", {
		BackgroundColor3 = T.Surface,
		Size = UDim2.new(1, 0, 1, -10),
		Parent = wrapper,
	})
	Corner(drop, 8)
	Stroke(drop, T.Stroke, 1, 0.3)
	local list = New("ScrollingFrame", {
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Size = UDim2.fromScale(1, 1),
		CanvasSize = UDim2.new(),
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
		ScrollBarThickness = 2,
		ScrollBarImageColor3 = T.Accent,
		Parent = drop,
	})
	Padding(list, 4, 4, 4, 4)
	List(list, 3)

	local obj = {
		Type = "Dropdown",
		Options = opts.Options or {},
		Multi = opts.Multi and true or false,
		Selected = {},
		Open = false,
	}
	local buttons = {}

	local function selectedArray()
		local result = {}
		for _, name in obj.Options do
			if obj.Selected[name] then
				table.insert(result, name)
			end
		end
		return result
	end

	function obj:Get()
		if self.Multi then
			return selectedArray()
		end
		return selectedArray()[1]
	end

	local function paint()
		for name, entry in buttons do
			local on = obj.Selected[name] == true
			Tween(entry.Button, 0.15, { BackgroundTransparency = on and 0.1 or 1 })
			entry.Label.TextColor3 = on and T.Text or T.SubText
			entry.Check.Instance.Visible = on and obj.Multi
		end
		local chosen = selectedArray()
		if #chosen == 0 then
			valueLabel.Text = opts.Placeholder or (obj.Multi and "None" or "Select...")
			valueLabel.TextColor3 = T.SubText
		else
			valueLabel.Text = table.concat(chosen, ", ")
			valueLabel.TextColor3 = T.Text
		end
	end

	local function setOpen(state)
		obj.Open = state
		local height = math.min(#obj.Options, 5) * 33 + 8 + 10
		if state then
			wrapper.Visible = true
			Tween(wrapper, 0.22, { Size = UDim2.new(1, 0, 0, height) })
			Tween(chevron.Instance, 0.22, { Rotation = 180 })
			Tween(boxStroke, 0.2, { Color = T.Accent })
		else
			Tween(wrapper, 0.18, { Size = UDim2.new(1, 0, 0, 0) })
			Tween(chevron.Instance, 0.18, { Rotation = 0 })
			Tween(boxStroke, 0.2, { Color = T.Stroke })
			task.delay(0.2, function()
				if not obj.Open and wrapper.Parent then
					wrapper.Visible = false
				end
			end)
		end
	end

	local function build()
		for _, entry in buttons do
			entry.Button:Destroy()
		end
		table.clear(buttons)

		for index, name in obj.Options do
			local button = New("TextButton", {
				Name = tostring(name),
				Text = "",
				AutoButtonColor = false,
				BackgroundColor3 = T.Accent,
				BackgroundTransparency = 1,
				Size = UDim2.new(1, 0, 0, 30),
				LayoutOrder = index,
				Parent = list,
			})
			Corner(button, 7)
			local label = New("TextLabel", {
				BackgroundTransparency = 1,
				Position = UDim2.fromOffset(10, 0),
				Size = UDim2.new(1, -36, 1, 0),
				Font = Fonts.Medium,
				TextSize = 13,
				TextColor3 = T.SubText,
				TextXAlignment = Enum.TextXAlignment.Left,
				TextTruncate = Enum.TextTruncate.AtEnd,
				Text = tostring(name),
				Parent = button,
			})
			local check = CreateIcon(button, "check", 14, T.Text)
			check.Instance.AnchorPoint = Vector2.new(1, 0.5)
			check.Instance.Position = UDim2.new(1, -10, 0.5, 0)
			check.Instance.Visible = false
			buttons[name] = { Button = button, Label = label, Check = check }

			button.MouseEnter:Connect(function()
				if not obj.Selected[name] then
					Tween(button, 0.12, { BackgroundTransparency = 0.8 })
				end
			end)
			button.MouseLeave:Connect(function()
				if not obj.Selected[name] then
					Tween(button, 0.12, { BackgroundTransparency = 1 })
				end
			end)
			button.Activated:Connect(function()
				if obj.Multi then
					obj.Selected[name] = not obj.Selected[name] or nil
				else
					obj.Selected = { [name] = true }
					setOpen(false)
				end
				paint()
				Sync(obj)
				Fire(opts.Callback, obj:Get())
			end)
		end
		paint()
	end

	function obj:Set(value, silent)
		self.Selected = {}
		if type(value) == "table" then
			for key, item in value do
				if type(key) == "number" then
					self.Selected[item] = true
				elseif item then
					self.Selected[key] = true
				end
			end
		elseif value ~= nil then
			self.Selected[value] = true
		end
		paint()
		Sync(self)
		if not silent then
			Fire(opts.Callback, self:Get())
		end
	end

	function obj:Refresh(newOptions, keepSelection)
		self.Options = newOptions or {}
		if not keepSelection then
			self.Selected = {}
		else
			for name in self.Selected do
				if not table.find(self.Options, name) then
					self.Selected[name] = nil
				end
			end
		end
		build()
		Sync(self)
		if self.Open then
			setOpen(true)
		end
	end

	click.Activated:Connect(function()
		setOpen(not obj.Open)
	end)

	build()
	if opts.Default ~= nil then
		obj:Set(opts.Default, true)
	end
	Register(obj, opts)
	return Finish(obj, row, title)
end

function Elements:Keybind(opts)
	opts = opts or {}
	local T = Library.Theme
	local row, header, title = NewRow(self, opts.Name or "Keybind", opts.Description, 92)

	local button = New("TextButton", {
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, 0, 0.5, 0),
		Size = UDim2.fromOffset(92, 30),
		BackgroundColor3 = T.Surface,
		AutoButtonColor = false,
		Font = Fonts.Bold,
		TextSize = 12,
		TextColor3 = T.Text,
		Text = "",
		ZIndex = 4,
		Parent = header,
	})
	Corner(button, 8)
	local buttonStroke = Stroke(button, T.Stroke, 1, 0.2)

	local obj = { Type = "Keybind", Value = opts.Default or Enum.KeyCode.Unknown, Listening = false }

	local function render()
		if obj.Listening then
			button.Text = "..."
			button.TextColor3 = T.AccentLight
		else
			button.Text = obj.Value == Enum.KeyCode.Unknown and "None" or obj.Value.Name
			button.TextColor3 = T.Text
		end
	end

	function obj:Set(key, silent)
		if typeof(key) == "string" then
			local ok, found = pcall(function()
				return Enum.KeyCode[key]
			end)
			key = ok and found or Enum.KeyCode.Unknown
		end
		self.Value = key or Enum.KeyCode.Unknown
		self.Listening = false
		render()
		Sync(self)
		if not silent then
			Fire(opts.OnChanged, self.Value)
		end
	end
	function obj:Get()
		return self.Value
	end

	button.Activated:Connect(function()
		obj.Listening = true
		Library._Listening = true
		Tween(buttonStroke, 0.15, { Color = T.AccentLight })
		render()
	end)

	Connect(UIS.InputBegan, function(input, processed)
		if obj.Listening then
			if input.UserInputType == Enum.UserInputType.Keyboard then
				Tween(buttonStroke, 0.15, { Color = T.Stroke })
				if input.KeyCode == Enum.KeyCode.Escape then
					obj:Set(Enum.KeyCode.Unknown)
				else
					obj:Set(input.KeyCode)
				end
				task.defer(function()
					Library._Listening = false
				end)
			end
			return
		end
		if processed or Library._Listening then
			return
		end
		if
			input.UserInputType == Enum.UserInputType.Keyboard
			and obj.Value ~= Enum.KeyCode.Unknown
			and input.KeyCode == obj.Value
		then
			Fire(opts.Callback, obj.Value)
		end
	end)

	obj.Value = opts.Default or Enum.KeyCode.Unknown
	render()
	Register(obj, opts)
	return Finish(obj, row, title)
end

local function MakeWindowDraggable(handle, target, gui, clamp, canDrag)
	local dragging, moved = false, false
	local dragStart, startPos

	handle.InputBegan:Connect(function(input)
		if IsPress(input) and (not canDrag or canDrag()) then
			dragging, moved = true, false
			dragStart = input.Position
			startPos = target.Position
		end
	end)
	Connect(UIS.InputChanged, function(input)
		if dragging and IsMove(input) then
			local delta = input.Position - dragStart
			if delta.Magnitude > 6 then
				moved = true
			end
			if moved then
				local x = startPos.X.Offset + delta.X
				local y = startPos.Y.Offset + delta.Y
				if clamp then
					local screen = gui.AbsoluteSize
					x = math.clamp(x, -screen.X / 2 + 30, screen.X / 2 - 30)
					y = math.clamp(y, -screen.Y / 2 + 30, screen.Y / 2 - 30)
				end
				target.Position = UDim2.new(startPos.X.Scale, x, startPos.Y.Scale, y)
			end
		end
	end)
	Connect(UIS.InputEnded, function(input)
		if dragging and IsPress(input) then
			dragging = false
		end
	end)
	return function()
		return moved
	end
end

				or string.format("%d:%02d", t // 60, t % 60)
			setStat("Uptime", text)
		end)
	)
	setStat("FPS", "--")
	setStat("Ping", "--")
	setStat("Players", #Players:GetPlayers() .. "/" .. Players.MaxPlayers)
	setStat("Memory", "--")

	end
	if opts.Side == "Left" or opts.Side == "Right" then
		widget.Side = opts.Side
	end
	widget:Apply()
	pop()

	self.StatsWidget = widget
	return widget
end

	local function looksLikeBackpack(scr, src)
		if scr then
			for frame in gameBackpacks do
				if
					scr:IsDescendantOf(frame)
					or scr:IsDescendantOf(frame:FindFirstAncestorOfClass("ScreenGui") or frame)
				then
					return true
				end
			end
			local full = scr:GetFullName():lower()
			if full:find("backpack", 1, true) or full:find("hotbar", 1, true) then
				return true
			end
		end
		if src and src ~= "" then
			if src:find("backpack", 1, true) or src:find("hotbar", 1, true) then
				return true
			end
			for _, name in guiNames() do
				if src:find(name, 1, true) then
					return true
				end
			end
		end
		return false
	end

	local function inGameBackpack(inst)
		for frame in gameBackpacks do
			if inst == frame or inst:IsDescendantOf(frame) then
				return true
			end
		end
		return false
	end

	local function referencesGameBackpack(fn)
		local getUps = (debug and debug.getupvalues) or getupvalues
		local getConsts = (debug and debug.getconstants) or getconstants
		local budget = 300

		local function scan(value, depth)
			budget -= 1
			if budget <= 0 then
				return false
			end
			if typeof(value) == "Instance" then
				return inGameBackpack(value)
			elseif type(value) == "table" and depth < 3 then
				for _, item in value do
					if scan(item, depth + 1) then
						return true
					end
					if budget <= 0 then
						return false
					end
				end
			end
			return false
		end

		if type(getUps) == "function" then
			local ok, ups = pcall(getUps, fn)
			if ok and type(ups) == "table" then
				for _, value in ups do
					local okScan, hit = pcall(scan, value, 0)
					if okScan and hit then
						return true
					end
				end
			end
		end
		if type(getConsts) == "function" then
			local ok, consts = pcall(getConsts, fn)
			if ok and type(consts) == "table" then
				local hotbar, equip, backpack = false, false, false
				for _, c in consts do
					if c == "Hotbar" then
						hotbar = true
					elseif c == "EquipTool" then
						equip = true
					elseif c == "Backpack" then
						backpack = true
					end
				end
				if hotbar or (equip and backpack) then
					return true
				end
			end
		end
		return false
	end

	local function muteGameInput()
		if typeof(getconnections) ~= "function" then
			return
		end
		local debugMode = env.Debug == true
		local ok, list = pcall(getconnections, UIS.InputBegan)
		if not ok or type(list) ~= "table" then
			return
		end
		for _, conn in list do
			if not mutedConns[conn] then
				local okFn, fn = pcall(function()
					return conn.Function
				end)
				local scr, src = nil, ""
				if okFn and type(fn) == "function" then
					scr = scriptOf(fn)
					local okSrc, chunk = pcall(function()
						return debug.info(fn, "s")
					end)
					if okSrc and type(chunk) == "string" then
						src = chunk:lower()
					end
				end
				local ours = scr ~= nil and isOurs(scr)
				local match = false
				if not ours then
					match = looksLikeBackpack(scr, src)
					if not match and okFn and type(fn) == "function" then
						match = referencesGameBackpack(fn)
					end
				end
				if debugMode then
					print(
						("[UI] InputBegan connection script=%s chunk=%s muted=%s"):format(
							scr and scr:GetFullName() or "?",
							src ~= "" and src or "?",
							tostring(match)
						)
					)
				end
				if match then
					local disabled = pcall(function()
						conn:Disable()
					end)
					if disabled then
						mutedConns[conn] = true
					end
				end
			end
		end
	end

	local function currentTools()
		local list = {}
		local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
		if backpack then
			for _, child in backpack:GetChildren() do
				if child:IsA("Tool") then
					table.insert(list, child)
				end
			end
		end
		local char = character()
		if char then
			for _, child in char:GetChildren() do
				if child:IsA("Tool") then
					table.insert(list, child)
				end
			end
		end
		table.sort(list, function(a, b)
			return (seen[a] or math.huge) < (seen[b] or math.huge)
		end)
		return list
	end

	refresh = function()
		for _, tool in currentTools() do
			if not seen[tool] then
				counter += 1
				seen[tool] = counter
				table.insert(
					toolConns,
					tool:GetPropertyChangedSignal("Name"):Connect(function()
						refresh()
					end)
				)
				table.insert(
					toolConns,
					tool:GetPropertyChangedSignal("TextureId"):Connect(function()
						refresh()
					end)
				)
			end
		end
		local tools = currentTools()
		local present = {}
		for _, tool in tools do
			present[tool] = true
		end

		for i = 1, 10 do
			if slots[i] and not present[slots[i]] then
				slots[i] = nil
			end
		end
		for tool, card in cards do
			if not present[tool] then
				card.Button:Destroy()
				cards[tool] = nil
				seen[tool] = nil
				unpinned[tool] = nil
			end
		end

		for _, tool in tools do
			if not slotOf(tool) and not unpinned[tool] then
				local free = firstFree()
				if free then
					slots[free] = tool
				end
			end
		end

		local char = character()

		for i = 1, 10 do
			local ui, tool = slotUI[i], slots[i]
			ui.Button.Visible = tool ~= nil
			if tool then
				paintIcon(ui.Image, ui.Label, tool)
				local equipped = tool.Parent == char
				Tween(ui.Button, 0.15, { BackgroundColor3 = equipped and T.Accent or T.Element })
				Tween(ui.Stroke, 0.15, { Color = equipped and T.AccentLight or T.Stroke })
				ui.Stroke.Thickness = equipped and 2 or 1
			end
		end

		local shown = 0
		for index, tool in tools do
			local card = cards[tool]
			if not card then
				card = makeCard(tool)
				cards[tool] = card
			end
			card.Button.LayoutOrder = index
			card.Button.Name = tool.Name
			paintIcon(card.Image, card.Label, tool)
			card.Caption.Text = tool.Name
			card.Caption.Visible = tool.TextureId ~= ""
			local slot = slotOf(tool)
			card.Badge.Visible = slot ~= nil
			card.Badge.Text = slot and (slot == 10 and "0" or tostring(slot)) or ""
			local equipped = tool.Parent == char
			card.Button.BackgroundColor3 = equipped and T.Accent or T.Element
			card.Stroke.Color = equipped and T.AccentLight or T.Stroke
			local match = filter == "" or tool.Name:lower():find(filter, 1, true) ~= nil
			card.Button.Visible = match
			if match then
				shown += 1
			end
		end
		emptyLabel.Visible = shown == 0
		emptyLabel.Text = #tools == 0 and "No tools" or "Nothing found"
		headLabel.Text = `Inventory`
	end

	local KEYS = {
		[Enum.KeyCode.One] = 1,
		[Enum.KeyCode.Two] = 2,
		[Enum.KeyCode.Three] = 3,
		[Enum.KeyCode.Four] = 4,
		[Enum.KeyCode.Five] = 5,
		[Enum.KeyCode.Six] = 6,
		[Enum.KeyCode.Seven] = 7,
		[Enum.KeyCode.Eight] = 8,
		[Enum.KeyCode.Nine] = 9,
		[Enum.KeyCode.Zero] = 10,
	}

	local CAS = game:GetService("ContextActionService")
	local lastKey, lastTime = nil, 0
	local pressToken = 0

	local function handleKey(keyCode)
		if Library._Listening then
			return false
		end
		if UIS:GetFocusedTextBox() then
			return false
		end
		if keyCode == lastKey and os.clock() - lastTime < 0.08 then
			return true
		end
		if keyCode == Enum.KeyCode.Backquote then
			lastKey, lastTime = keyCode, os.clock()
			bp:ToggleInventory()
			return true
		end
		local index = KEYS[keyCode]
		if index then
			lastKey, lastTime = keyCode, os.clock()
			local tool = slots[index]
			if tool then
				pressToken += 1
				local token = pressToken
				local want = tool.Parent ~= character()
				toggleEquip(tool)
				showTag(tool.ToolTip ~= "" and tool.ToolTip or tool.Name, 1.5)

				task.spawn(function()
					for _, delay in { 0, 0.05, 0.15 } do
						task.wait(delay)
						if not alive or token ~= pressToken then
							return
						end
						local hum, char = humanoid(), character()
						if not hum or not char or not tool.Parent then
							return
						end
						if want then
							if tool.Parent ~= char then
								hum:EquipTool(tool)
							end
						elseif char:FindFirstChildOfClass("Tool") then
							hum:UnequipTools()
						end
					end
				end)
			end
			return true
		end
		return false
	end

	local keyList = { Enum.KeyCode.Backquote }
	for keyCode in KEYS do
		table.insert(keyList, keyCode)
	end
	pcall(function()
		CAS:BindActionAtPriority("WineBackpackKeys", function(_, state, input)
			if state ~= Enum.UserInputState.Begin then
				return Enum.ContextActionResult.Pass
			end
			if handleKey(input.KeyCode) then
				return Enum.ContextActionResult.Sink
			end
			return Enum.ContextActionResult.Pass
		end, false, Enum.ContextActionPriority.High.Value + 100, table.unpack(keyList))
	end)

	On(UIS.InputBegan, function(input)
		if input.UserInputType == Enum.UserInputType.Keyboard then
			handleKey(input.KeyCode)
		end
	end)

	function bp:Refresh()
		refresh()
	end

	function bp:Destroy()
		for _, c in conns do
			pcall(function()
				c:Disconnect()
			end)
		end
		table.clear(conns)
		for _, c in toolConns do
			pcall(function()
				c:Disconnect()
			end)
		end
		table.clear(toolConns)
		for _, inst in { hotbar, nameTag, panel } do
			pcall(function()
				inst:Destroy()
			end)
		end
		alive = false
		pcall(function()
			game:GetService("ContextActionService"):UnbindAction("WineBackpackKeys")
		end)
		for inst, info in suppressed do
			pcall(function()
				inst[info.Prop] = info.Value
			end)
		end
		for conn in mutedConns do
			pcall(function()
				conn:Enable()
			end)
		end

		for frame, visible in nativeVisible do
			pcall(function()
				if wasEnabled then
					frame.Visible = visible
				end
			end)
		end
		pcall(function()
			StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.Backpack, wasEnabled)
		end)
		if Library.Backpack == self then
			Library.Backpack = nil
		end
		if Library.StatsWidget then
			Library.StatsWidget:Apply()
		end
	end

	self.Backpack = bp
	refresh()
	if self.StatsWidget then
		self.StatsWidget:Apply()
	end
	return bp
end

function Library:DisableBackpack()
	if self.Backpack then
		self.Backpack:Destroy()
	end
end

	local function ApplyLayout(animate)
		local screen = gui.AbsoluteSize
		if screen.X < 10 then
			local camera = workspace.CurrentCamera
			screen = camera and camera.ViewportSize or Vector2.new(800, 600)
		end
		local width, height
		if fullscreen then
			width = math.max(screen.X - 24, 300)
			height = math.max(screen.Y - 24, 220)
		else
			width = math.clamp(screen.X - 24, 300, opts.Width or 600)
			height = math.clamp(screen.Y - 24, 220, opts.Height or 420)
		end
		local target = UDim2.fromOffset(width, height)
		if animate then
			Tween(main, 0.45, { Size = target }, Enum.EasingStyle.Quint)
		else
			main.Size = target
		end

		compact = IsMobile or width < 520
		local sideW = compact and 58 or 164
		sidebar.Size = UDim2.new(0, sideW, 1, -topH)
		content.Position = UDim2.fromOffset(sideW, topH)
		content.Size = UDim2.new(1, -sideW, 1, -topH)
		for _, tab in Window.Tabs do
			tab:_ApplyCompact(compact)
		end
	end

	function Window:SelectTab(tab)
		for _, other in self.Tabs do
			local on = other == tab
			other.Page.Visible = on
			Tween(other.Button, 0.18, { BackgroundTransparency = on and 0 or 1 })
			Tween(other._Indicator, 0.18, { BackgroundTransparency = on and 0 or 1 })
			other._Icon:SetColor(on and T.AccentLight or T.SubText, 0.18)
			Tween(other._Label, 0.18, { TextColor3 = on and T.Text or T.SubText })
		end
		self.SelectedTab = tab
		tab.Page.Position = UDim2.fromOffset(0, 14)
		Tween(tab.Page, 0.25, { Position = UDim2.new() })
	end

	local fabIcon = nil
	local firstClose = true

  local rest = main.Position
	local settled = false
	local animToken = 0
	local SLIDE = IsMobile and 18 or 28

	function Window:Toggle(state)
		if state == nil then
			state = not self.Visible
		end
		self.Visible = state
		if fabIcon then
			fabIcon:Set(state and "x-bold" or "menu")
		end

		animToken += 1
		local token = animToken

		if state then
			if not main.Visible then
				main.GroupTransparency = 1
				scale.Scale = 0.96
				main.Position = rest + UDim2.fromOffset(0, SLIDE)
			end
			main.Visible = true
			settled = false
			Tween(main, 0.35, { GroupTransparency = 0 }, Enum.EasingStyle.Quad)
			Tween(main, 0.55, { Position = rest }, Enum.EasingStyle.Quint)
			Tween(scale, 0.55, { Scale = 1 }, Enum.EasingStyle.Quint)
			task.delay(0.55, function()
				if token == animToken then
					settled = true
				end
			end)
		else
			if settled then
				rest = main.Position
			end
			settled = false
			Tween(main, 0.26, { GroupTransparency = 1 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
			Tween(
				main,
				0.3,
				{ Position = rest + UDim2.fromOffset(0, SLIDE * 0.6) },
				Enum.EasingStyle.Quint,
				Enum.EasingDirection.In
			)
			Tween(scale, 0.3, { Scale = 0.97 }, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
			task.delay(0.3, function()
				if token == animToken then
					main.Visible = false
					main.Position = rest
				end
			end)
			if firstClose and not IsMobile then
				firstClose = false
				self:Notify({
					Title = "UI hidden",
					Content = `Press {self.ToggleKey.Name} to open it again.`,
					Icon = "keyboard",
					Duration = 3,
				})
			end
		end
	end

	local savedPos = nil
	function Window:SetFullscreen(state)
		state = state and true or false
		if state == fullscreen then
			return
		end
		if state then
			savedPos = self.Visible and main.Position or rest
			rest = UDim2.fromScale(0.5, 0.5)
		else
			rest = savedPos or UDim2.fromScale(0.5, 0.5)
		end
		fullscreen = state
		fullscreenIcon:Set(state and "minimize" or "maximize")
		if fullscreenFallback then
			fullscreenFallback.Text = state and "[-]" or "[]"
		end
		ApplyLayout(true)
		if main.Visible then
			Tween(main, 0.45, { Position = rest }, Enum.EasingStyle.Quint)
		end
	end

	function Window:ToggleFullscreen()
		self:SetFullscreen(not fullscreen)
	end

function Library:Destroy()
	if self._Destroyed then
		return
	end
	self._Destroyed = true

	for _, toggle in table.clone(self._Toggles) do
		KillToggle(toggle)
	end
	table.clear(self._Toggles)

	for _, hook in self._UnloadHooks do
		task.spawn(function()
			local ok, err = pcall(hook)
			if not ok then
				warn("[UI] unload hook error: " .. tostring(err))
			end
		end)
	end
	table.clear(self._UnloadHooks)

	if self.Backpack then
		pcall(function()
			self.Backpack:Destroy()
		end)
	end
	if self.StatsWidget then
		pcall(function()
			self.StatsWidget:Destroy()
		end)
	end

	SaveIconMeta()

	for _, connection in self._Connections do
		pcall(function()
			connection:Disconnect()
		end)
	end
	table.clear(self._Connections)
	if self._Gui then
		self._Gui:Destroy()
		self._Gui = nil
	end
	table.clear(self.Flags)
	table.clear(self.Options)
	if env.Mal == self then
		env.Mal = nil
	end
end

env.Mal = Library
return Library
