local function ntf(a, b)
    pcall(function()
        game:GetService("StarterGui"):SetCore("SendNotification", {Title = tostring(a), Text = tostring(b), Duration = 3})
    end)
end

local function fpr(p)
    if not (p and p:IsA("ProximityPrompt")) then return end
    if fireproximityprompt then if pcall(fireproximityprompt, p) then return end end
    pcall(function()
        local h, d, l = p.HoldDuration, p.MaxActivationDistance, p.RequiresLineOfSight
        p.HoldDuration = 0 p.MaxActivationDistance = 1000 p.RequiresLineOfSight = false
        p:InputHoldBegin() task.wait() p:InputHoldEnd()
        p.MaxActivationDistance = d p.HoldDuration = h p.RequiresLineOfSight = l
    end)
end

local function oName(o)
    local v = ""
    pcall(function()
        if o:IsA("StringValue") then v = tostring(o.Value) else v = o.Value and o.Value.Name or "" end
    end)
    return v
end

local function findMyHouse()
    for _, o in ipairs(workspace:GetDescendants()) do
        if o.Name:lower() == "owner" and (o:IsA("StringValue") or o:IsA("ObjectValue")) then
            local n = oName(o)
            if n == LP.Name or n == LP.DisplayName then return o.Parent end
        end
    end
    return nil
end

local function getSafe()
    for _, o in ipairs(workspace:GetDescendants()) do
        if o.Name:lower() == "owner" and (o:IsA("StringValue") or o:IsA("ObjectValue")) then
            local n = oName(o)
            if n == LP.Name or n == LP.DisplayName then
                local a = o.Parent
                while a and a ~= workspace do
                    local s = a:FindFirstChild("Safe", true)
                    if s then
                        local p = s:IsA("BasePart") and s or s:FindFirstChildWhichIsA("BasePart", true)
                        return s, p and p.CFrame
                    end
                    a = a.Parent
                end
            end
        end
    end
    return nil, nil
end

local function GetLockedTools()
    local items = {}
    local char = LP.Character
    local bp = LP:FindFirstChild("Backpack")
    if char then for _, t in pairs(char:GetChildren()) do
        if t:IsA("Tool") and not dupeBlacklist[t.Name] and not table.find(items, t.Name) then table.insert(items, t.Name) end
    end end
    if bp then for _, t in pairs(bp:GetChildren()) do
        if t:IsA("Tool") and not dupeBlacklist[t.Name] and not table.find(items, t.Name) then table.insert(items, t.Name) end
    end end
    table.sort(items); return items
end

local function dupe1(n)
    local s, cf = getSafe()
    if not s or not cf then return false end
    local c = LP.Character
    local r = c and c:FindFirstChild("HumanoidRootPart")
    if not r then return false end
    local o = r.CFrame
    pcall(function()
        r.CFrame = cf * CFrame.new(0, 2, 0) task.wait(0.5)
        for _, pr in ipairs(s:GetDescendants()) do
            if pr:IsA("ProximityPrompt") and pr.Enabled then
                local od = pr.HoldDuration
                pr.HoldDuration = 0 pr.RequiresLineOfSight = false
                pcall(function() fpr(pr) end)
                pr.HoldDuration = od
            end
        end
        task.wait(0.5)
        task.spawn(function() RS:WaitForChild("BackpackRemote"):InvokeServer("Store", n) end)
        task.spawn(function() RS:WaitForChild("Inventory"):FireServer("Change", n, "Backpack", s) end)
        task.wait(0.8) r.CFrame = o task.wait(0.8) RS:WaitForChild("BackpackRemote"):InvokeServer("Grab", n) task.wait(0.5)
    end)
    return true
end

-- DRAGGING (smooth follow + tilt + bounce)
local dragging = false
local dragInput = nil
local dragStart, startPos
local targetPos = Main.Position

local NORMAL_SCALE = 0.62
local DRAG_SCALE = 0.67

local function updateListPos()
    if listOpen and ListFrame.Visible then
        ListFrame.Position = UDim2.fromOffset(DropBtn.AbsolutePosition.X, DropBtn.AbsolutePosition.Y + DropBtn.AbsoluteSize.Y + 6)
    end
end

TitleBar.InputBegan:Connect(function(input)
    if locked then return end
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
        targetPos = Main.Position
        TweenService:Create(UIScale, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = DRAG_SCALE}):Play()
        TweenService:Create(Main, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Rotation = -1}):Play()
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
                TweenService:Create(UIScale, TweenInfo.new(0.45, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = NORMAL_SCALE}):Play()
                TweenService:Create(Main, TweenInfo.new(0.4, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), {Rotation = 0}):Play()
            end
        end)
    end
end)

TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UIS.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        delta = delta / UIScale.Scale -- cursor moves 1:1 even while scaled down
        targetPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

RunService.RenderStepped:Connect(function(dt)
    if not dragging then return end
    local lerp = math.clamp(dt * 16, 0, 1)
    local newX = Main.Position.X.Offset + (targetPos.X.Offset - Main.Position.X.Offset) * lerp
    local newY = Main.Position.Y.Offset + (targetPos.Y.Offset - Main.Position.Y.Offset) * lerp
    local velX = targetPos.X.Offset - Main.Position.X.Offset
    Main.Position = UDim2.new(Main.Position.X.Scale, newX, Main.Position.Y.Scale, newY)
    Main.Rotation = math.clamp(velX * 0.012, -4, 4) -- tilt while whipping around
    updateListPos()
end)

    b.Size = UDim2.new(1, 0, 0, 24)
    b.BackgroundColor3 = Color3.fromRGB(18, 18, 20)
    b.BackgroundTransparency = 0.25
    b.BorderSizePixel = 0
    b.Text = text
    b.Font = Enum.Font.GothamBold
    b.TextSize = 11
    b.TextColor3 = Color3.fromRGB(255, 255, 255)
    b.AutoButtonColor = false
    Instance.new("UICorner", b).CornerRadius = UDim.new(0, 6)
    b.Parent = Scroll
    return b
end

local function highlight()
    for name, b in pairs(listItems) do
        if b and b.Parent then
            b.TextColor3 = (name == (selectedDupeItem or "ALL (RANDOM)")) and Color3.fromRGB(235, 20, 26) or Color3.fromRGB(255, 255, 255)
        end
    end
end

local function rebuildList()
    for _, b in pairs(listItems) do if b then b:Destroy() end end
    table.clear(listItems)
    listItems["ALL (RANDOM)"] = makeEntry("ALL (RANDOM)", 1)
    listItems["ALL (RANDOM)"].MouseButton1Click:Connect(function()
        selectedDupeItem = nil
        highlight()
        closeList()
        setStatus("Item: ALL (random)")
        ntf("Auto Dupe", "duping: ALL (random)")
    end)
    local order = 2
    for _, name in ipairs(GetLockedTools()) do
        local n = name
        listItems[n] = makeEntry(n, order)
        listItems[n].MouseButton1Click:Connect(function()
            selectedDupeItem = n
            highlight()
            closeList()
            setStatus("Item: " .. n)
            ntf("Auto Dupe", "duping: " .. n)
        end)
        order = order + 1
    end
    local refresh = makeEntry("⟳  REFRESH", order)
    refresh.TextColor3 = Color3.fromRGB(155, 155, 155)
    refresh.MouseButton1Click:Connect(function()
        rebuildList()
        highlight()
        ntf("Auto Dupe", "refreshed: " .. tostring(#GetLockedTools()) .. " items")
    end)
    highlight()
end

closeList = function()
    listOpen = false
    ListFrame.Visible = false
    DropBtn.Text = (selectedDupeItem or "ALL (RANDOM)") .. "  ▼"
end

DropBtn.MouseButton1Click:Connect(function()
    if listOpen then
        closeList()
        return
    end
    rebuildList()
    listOpen = true
    local count = 0
    for _ in pairs(listItems) do count = count + 1 end
    local w = math.max(DropBtn.AbsoluteSize.X, 180)
    ListFrame.Position = UDim2.fromOffset(DropBtn.AbsolutePosition.X, DropBtn.AbsolutePosition.Y + DropBtn.AbsoluteSize.Y + 6)
    ListFrame.Size = UDim2.fromOffset(w, math.clamp(count * 28 + 12, 80, 220))
    ListFrame.Visible = true
    DropBtn.Text = (selectedDupeItem or "ALL (RANDOM)") .. "  ▲"
end)

-- main toggle
Hit.MouseButton1Click:Connect(function()
    if dupeOn then
        dupeOn = false
        toggleSwitch(Track, Knob, false)
        setStatus("Stopped")
        setProgress(0, "READY")
        ntf("Auto Dupe", "stopped")
        return
    end
    if not findMyHouse() then
        ntf("Auto Dupe", "no owned house / safe found")
        setStatus("No house found")
        return
    end
    dupeOn = true
    toggleSwitch(Track, Knob, true)
    setStatus("Running - " .. (selectedDupeItem or "ALL (random)"))
    setProgress(0.1, "STARTING")
    ntf("Auto Dupe", "started - " .. (selectedDupeItem or "ALL (random)"))
    task.spawn(function()
        while dupeOn do
            local t = GetLockedTools()
            if #t == 0 then
                setStatus("No duplicable items")
                task.wait(1)
            else
                if not getSafe() then
                    ntf("Auto Dupe", "lost safe - stopped")
                    setStatus("Lost safe - stopped")
                    dupeOn = false
                    toggleSwitch(Track, Knob, false)
                    setProgress(0, "READY")
                    break
                end
                local pick = selectedDupeItem and table.find(t, selectedDupeItem) and selectedDupeItem
                    or t[math.random(1, #t)]
                setStatus("Duping: " .. pick)
                setProgress(0.5, "DUPING")
                dupe1(pick)
                setProgress(1, "DONE")
                task.wait(0.15)
                if dupeOn then setProgress(0.1, "NEXT") end
            end
        end
    end)
end)
